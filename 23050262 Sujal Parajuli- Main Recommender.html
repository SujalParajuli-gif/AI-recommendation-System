User-based Item-CF Top-N function¶

Generating recommendations by aggregating similarity scores from the user's history.

In [58]:
def recommend_itemcf_user(user_id, N=10):
    # Checking if the user exists in the interaction matrix
    if user_id not in interaction_matrix.index:
        return None  # returning None for an unknown user

    # Getting the user's interaction/rating row
    user_row = interaction_matrix.loc[user_id]

    # Selecting only the items the user has interacted with (values > 0)
    hist = user_row[user_row > 0]
    if hist.empty:
        return None  # returning None for cold-start users (no history)

    # Getting the ASINs the user has interacted with
    hist_asins = hist.index.tolist()

    # Converting those ASINs into their column indexes (for using the similarity matrix)
    hist_idx = [item_index[a] for a in hist_asins if a in item_index]

    # Taking the user's interaction values as weights (how strong the user liked/interacted)
    weights = hist.values.astype(np.float32)

    # Scoring all items by combining similarity to the user's history items (weighted sum)
    scores = sim_matrix[:, hist_idx] @ weights  # generating item scores

    # Removing items the user has already seen by setting their scores to very low
    scores[hist_idx] = -np.inf  # removing seen items

    # Selecting the indexes of the top N highest-scoring items
    top_idx = np.argsort(scores)[::-1][:N]

    # Returning the recommended item ASINs
    return [items[i] for i in top_idx]

Model 3: TruncatedSVD (latent factor model)¶

Training TruncatedSVD on the interaction matrix and reconstructing predicted scores.

In [59]:
# Creating the SVD model for reducing the interaction matrix into 20 hidden features
svd = TruncatedSVD(n_components=20, random_state=42)

# Fitting the SVD model and transforming the user-item matrix into a latent (compressed) representation
latent_matrix = svd.fit_transform(interaction_matrix.values)

# Reconstructing the full user-item score matrix (predicted scores for all users and items)
reconstructed = latent_matrix @ svd.components_

def recommend_svd_user(user_id, N=10):
    # Handling unknown users (user not present in the interaction matrix)
    if user_id not in interaction_matrix.index:
        return None

    # Getting the row position (index) of the user inside the interaction matrix
    uidx = interaction_matrix.index.get_loc(user_id)

    # Getting the predicted scores for this user from the reconstructed matrix
    scores = reconstructed[uidx].copy()

    # Finding which items the user has already interacted with (seen items)
    seen = interaction_matrix.iloc[uidx].values > 0

    # Removing already-seen items by setting their scores to very low
    scores[seen] = -np.inf

    # Getting the indexes of the top N highest-scoring items
    top_idx = np.argsort(scores)[::-1][:N]

    # Returning the recommended item ASINs
    return [items[i] for i in top_idx]

# Precision@K: how many of the top K recommendations are correct
def precision_at_k(recs, rel_set, k=10):
    return len(set(recs[:k]) & rel_set) / k if recs else 0.0

# Recall@K: how many of the correct items we managed to recommend in top K
def recall_at_k(recs, rel_set, k=10):
    return len(set(recs[:k]) & rel_set) / len(rel_set) if rel_set else 0.0

# NDCG@K: checks ranking quality (correct items higher in the list = better score)
def ndcg_at_k(recs, rel_set, k=10):
    # DCG: giving more credit when a relevant item appears earlier in the list
    dcg = sum((1.0 if a in rel_set else 0.0) / math.log2(i + 1)
              for i, a in enumerate(recs[:k], start=1))
    
    # IDCG: best possible DCG (ideal ranking) used for normalization
    ideal_hits = min(len(rel_set), k)
    idcg = sum(1.0 / math.log2(i + 1) for i in range(1, ideal_hits + 1))
    
    return dcg / idcg if idcg > 0 else 0.0

# Evaluating a recommendation method by averaging metrics over many test users
def evaluate_method(get_recs_fn, k=10, max_users=500):
    metrics = []

    # Looping through test data (one held-out item per user) and limiting to max_users
    for user_id, true_asin in test_df[["user_id", "parent_asin"]].values[:max_users]:
        
        # Skipping if the user does not exist in the training interaction matrix
        if user_id not in interaction_matrix.index:
            continue

        # Getting recommendations from the model (or using popularity as fallback)
        recs = get_recs_fn(user_id, k) or recommend_popularity(user_id, k)

        # Treating the held-out test item as the only "relevant" item
        rel = {true_asin}

        # Saving the three metric values for this user
        metrics.append([
            precision_at_k(recs, rel, k),
            recall_at_k(recs, rel, k),
            ndcg_at_k(recs, rel, k),
        ])

    # Returning the average of each metric across all evaluated users
    return pd.Series(
        np.mean(metrics, axis=0),
        index=["Precision@10", "Recall@10", "NDCG@10"]
    )

# Comparing all three models and making a final results table
results = pd.DataFrame({

# Implementing a simple notebook CLI for user-based Top-N recommendations
def run_cli():
    while True:
        # Asking the user to choose between generating recommendations or quitting
        choice = input("\nChoose: 1) Recommend for user  2) Quit: ").strip()

        # Quitting the CLI loop when option 2 is selected
        if choice == "2":
            print("Exited")
            break

        # Handling invalid menu options
        if choice != "1":
            print("Invalid choice. Type 1 or 2.")
            continue

        # Taking user_id input for generating personalized recommendations
        user_id = input("Enter user_id: ").strip()

        # Generating recommendations using Item–Item similarity (ItemCF)
        itemcf = recommend_itemcf_user(user_id, 10)

        # Generating recommendations using TruncatedSVD (latent factor model)
        svdrec = recommend_svd_user(user_id, 10)

        # Handling cold-start or unknown user by falling back to popularity baseline
        if (user_id not in interaction_matrix.index) or (itemcf is None and svdrec is None):
            print("\nNo interaction history or unknown user. Recommending Top-N popular items (Popularity baseline).")
            pop = recommend_popularity(user_id, 10)
            print_recs("Popularity baseline (Top 10):", pop)
            continue

        # Displaying Item–Item similarity recommendations when available
        if itemcf is None:
            print("\nItem–Item similarity: Cold-start (no history available).")
        else:
            print_recs("Item–Item similarity (Top 10):", itemcf)

        # Displaying TruncatedSVD recommendations when available
        if svdrec is None:
            print("\nTruncatedSVD: Unknown user_id (not in training data).")
        else:
            print_recs("TruncatedSVD (Top 10):", svdrec)
